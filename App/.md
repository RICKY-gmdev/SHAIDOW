graph TD
    %% User
    A[User types message and taps 'Send']

    %% .NET MAUI Frontend (Client)
    B[OnSendMessage event is triggered]
    C[Create User & AI Bubbles]
    D[User's message is added to ChatList]
    E[An EMPTY AI message bubble is added to ChatList]
    F[ApiService.StreamChatResponseAsync is called]
    G[Makes POST request to /chat with message & thread_id]
    H["Receives SSE events (text_chunk, tool_start, etc.)"]
    I[Updates the AI message bubble in real-time on the UI thread]
    J[On 'stream_end', hide spinner]

    %% FastAPI Backend (Server)
    K["/chat endpoint receives request"]
L["Calls LangGraph agent_executor.astream_events"]
M["event_stream() function catches agent events"]
N["Formats events into JSON (e.g., data: {type: text_chunk, ...})"]
T["Tool Node: Executes the required tool (e.g., search_for_image_tool)"]
V["Tool returns result (e.g., IMAGE_URL::...)"]


    %% LangGraph Agent (Core Logic)
    P[Graph starts: Preprocesses input]
    Q[Agent Node: Invokes Gemini LLM with tools]
    R{Decision: Tool needed?}
    S[LLM generates final text response]
    T["Tool Node: Executes the required tool (e.g., search_for_image_tool)"]
    U[Calls External API]
    V["Tool returns result (e.g., IMAGE_URL::...)"]
    W[Graph execution finishes for this turn]

    %% External Tools & APIs
    X[Pexels API]
    Y[Stability AI API]
    Z[Claude/Mistral APIs]

    %% Flow connections
    A --> B
    B --> C
    C --> D
    C --> E
    E --> F
    F --> G
    G -- HTTP Request --> K
    G -- SSE Stream --> H
    H --> I
    I --> J

    K --> L
    L -- Yields Events --> M
    M --> N
    N --> O
    O -- HTTP Response Stream --> H

    P --> Q
    Q --> R
    R -- No --> S
    R -- Yes --> T
    T --> U
    U --> V
    V --> Q
    S --> W

    T -- Generates on_tool_start / on_tool_end --> M
    Q -- Generates on_chat_model_stream --> M

    U -- Chooses API --> X
    U -- Chooses API --> Y
    U -- Chooses API --> Z
